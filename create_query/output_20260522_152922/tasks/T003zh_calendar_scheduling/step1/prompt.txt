      type: object

          description: 事件ID
      required: [event_id]

  - name: calendar_create_event

      type: object

          type: string
          description: 结束时间 (ISO 8601)

            type: string
          description: 参加者列表

          type: string
          description: 地点
      required: [title, start_time, end_time]

  - name: calendar_get_user_events

      type: object
      properties:

          type: string
          description: 查询日期 (YYYY-MM-DD)
      required: [user, date]

  - name: calendar_delete_event
    description: 删除日历事件
    input_schema:
      type: object
      properties:
        event_id:
          type: string
          description: 事件ID
      required: [event_id]
